add_library(greatest INTERFACE)
# SYSTEM so that warnings from the test framework do not fail a -Werror build.
target_include_directories(greatest SYSTEM INTERFACE greatest)

function(tcli_add_test name)
    add_executable(${name} ${name}.c)
    target_link_libraries(${name} PRIVATE tcli::tcli greatest)
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wshadow -Wundef)
        if(TCLI_WERROR)
            target_compile_options(${name} PRIVATE -Werror)
        endif()
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()
